# -*- coding: utf-8 -*-
import io, re, sys, glob, xml.etree.ElementTree as ET
sys.path.insert(0, ".")
import modules.task12_gift as M

B = re.compile(r"\[(-?\d+),(-?\d+)\]\[(-?\d+),(-?\d+)\]")

def load(path):
    s = io.open(path, encoding="utf-8", errors="replace").read()
    root = ET.fromstring(s.encode("utf-8"))
    out = []
    for n in root.iter():
        m = B.match(n.get("bounds") or "")
        if not m:
            continue
        x1, y1, x2, y2 = [int(x) for x in m.groups()]
        if x2 <= x1 or y2 <= y1:
            continue
        out.append({
            "text": (n.get("text") or "").strip(),
            "desc": (n.get("content-desc") or "").strip(),
            "bounds": (x1, y1, x2, y2),
            "clickable": n.get("clickable") == "true",
            "class": (n.get("class") or "").rsplit(".", 1)[-1],
        })
    return out

for p in sorted(glob.glob("_probe_gift_*.xml")):
    ns = load(p)
    rows = M.parse_task_rows(ns)
    print("\n=== %s  nodes=%d rows=%d" % (p, len(ns), len(rows)))
    print("    flags: home_tab=%s gift_tab=%s chou=%s earn=%s new=%s gift=%s wish=%s panel=%s" % (
        M.find_new_tab(ns), M.find_gift_tab(ns), M.find_chou_button(ns),
        M.find_earn_button(ns), M.is_new_page(ns), M.is_gift_page(ns),
        M.is_wish_page(ns), M.is_panel(ns)))
    for r in rows:
        print("    X=%-3d cy=%-5d btn=%-5s key=%-30s sub=%s" % (
            r["secs"], r["cy"], r["btn"], r["key"], r["sub"]))
