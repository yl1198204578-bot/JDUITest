# -*- coding: utf-8 -*-
import io, re, sys, xml.etree.ElementTree as ET
sys.path.insert(0, ".")
import modules.task12_gift as M

B = re.compile(r"\[(-?\d+),(-?\d+)\]\[(-?\d+),(-?\d+)\]")

def load(path, keep_bad=False):
    s = io.open(path, encoding="utf-8", errors="replace").read()
    root = ET.fromstring(s.encode("utf-8"))
    out = []
    for n in root.iter():
        m = B.match(n.get("bounds") or "")
        if not m:
            continue
        b = tuple(int(x) for x in m.groups())
        if not keep_bad and (b[2] <= b[0] or b[3] <= b[1]):
            continue
        out.append({
            "text": (n.get("text") or "").strip(),
            "desc": (n.get("content-desc") or "").strip(),
            "bounds": b,
            "clickable": n.get("clickable") == "true",
            "class": (n.get("class") or "").rsplit(".", 1)[-1],
            "displayed": n.get("displayed"),
        })
    return out

PANEL = "_probe_gift_panel.xml"
ns = load(PANEL)
print("=== panel nodes=%d ===" % len(ns))
print("--- 所有含 浏览 / 许愿 / 次数 / 助力 / 下单 的文本节点 ---")
for n in ns:
    t = n["text"]
    if any(k in t for k in (u"浏览", u"许愿", u"次数", u"助力", u"下单", u"完成", u"领取")):
        print("  %-46s %s clickable=%s" % (t[:46], n["bounds"], n["clickable"]))
print("--- 右侧按钮候选 (x1>=700) ---")
for n in ns:
    if n["bounds"][0] >= 700 and n["text"] and len(n["text"]) <= 8:
        print("  %-12s %s clickable=%s" % (n["text"], n["bounds"], n["clickable"]))
print("--- y<790 的有效 bounds 文本（SAFE_TOP 陷阱） ---")
for n in ns:
    b = n["bounds"]
    if b[3] < 790 and n["text"]:
        print("  %-30s %s" % (n["text"][:30], b))

for p in ("_probe_gift_home.xml", "_probe_gift_newtab_full.xml",
          "_probe_gift_gift_tab.xml", "_probe_gift_after_chou.xml",
          "_probe_gift_landing.xml"):
    ns = load(p)
    print("\n=== %s ===" % p)
    for n in ns:
        t = n["text"]
        if t in (u"新品", u"送礼新品", u"京东新品", u"品质好礼逛京东",
                 u"好礼许愿机", u"赚次数", u"做任务获得许愿次数", u"image"):
            print("  %-24s %-28s clickable=%s class=%s" % (
                t, n["bounds"], n["clickable"], n["class"]))
