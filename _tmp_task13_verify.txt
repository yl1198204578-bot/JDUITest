# -*- coding: utf-8 -*-
import io
import os
import re
import sys
import xml.etree.ElementTree as ET

sys.path.insert(0, ".")
import modules.task13_female as M

B = re.compile(r"\[(-?\d+),(-?\d+)\]\[(-?\d+),(-?\d+)\]")


def load(path):
    root = ET.fromstring(io.open(path, encoding="utf-8", errors="replace").read().encode("utf-8"))
    out = []
    for n in root.iter():
        m = B.match(n.get("bounds") or "")
        if not m:
            continue
        b = tuple(int(x) for x in m.groups())
        if b[2] <= b[0] or b[3] <= b[1]:
            continue
        out.append({
            "text": (n.get("text") or "").strip(),
            "desc": (n.get("content-desc") or "").strip(),
            "bounds": b,
            "clickable": n.get("clickable") == "true",
            "class": (n.get("class") or "").rsplit(".", 1)[-1],
        })
    return out


TAGS = ["home", "home2", "kk2", "kk2c", "channels", "channels2", "chan",
        "chan_qy", "chan_qy2", "female_ok", "female_ok2", "qy_female",
        "qy_female2", "female_page2", "panel_a", "panel_b", "panel_top",
        "panel_cur", "back_panel", "claim1", "closed", "r0", "r1", "r2",
        "r3", "r4", "t0", "t4", "t14", "landing", "s2", "s3"]

for tag in TAGS:
    p = "_probe_female_%s.xml" % tag
    if not os.path.isfile(p):
        print("%-16s MISSING" % tag)
        continue
    ns = load(p)
    st = M.classify(ns)
    tasks = M.parse_rows(ns, "task")
    claims = M.parse_rows(ns, "claim")
    print("%-16s nodes=%3d state=%-9s tasks=%d claims=%d sig=%d" %
          (tag, len(ns), st, len(tasks), len(claims), len(M.panel_sig(ns))))
    if tasks:
        for r in tasks[:8]:
            print("      T y=%4d btn=%-5s title=%r" %
                  (r["bounds"][1], r["text"], r["title"]))
    if claims:
        for r in claims[:4]:
            print("      C y=%4d title=%r" % (r["bounds"][1], r["title"]))

print("=== r3==r4 sig:", M.panel_sig(load("_probe_female_r3.xml")) ==
      M.panel_sig(load("_probe_female_r4.xml")))
print("=== r0==r1 sig:", M.panel_sig(load("_probe_female_r0.xml")) ==
      M.panel_sig(load("_probe_female_r1.xml")))
