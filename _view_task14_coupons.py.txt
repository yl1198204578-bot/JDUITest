  → 页面里【天天领豆 / 最高1000京豆 / 领更多】卡片（XML 里节点叫【更多任务】）
  → 点【领更多】弹出【做任务得奖励】任务面板

  「SK-II洗面奶13.9元限时抢」等一律排除，哪怕按钮也是【去完成】。

  循环扫描到整张任务列表没有可做任务为止。

真机踩坑（2026-11-13 实测，夹具见 _fixture_coupon_*）：
  * 首次进【领券中心】常弹「恭喜获得 ¥15 满30元可用」浮层；

  * 弹窗叠在【领券中心】之上，XML 里两页节点混在一起；滚动后被弹窗顶部

# ---------------- 纯函数：解析与判定 ----------------
def norm(t):
    """H5 会把文字渲染成竖排 / OCR 带噪声，关键字匹配前压掉空白。"""
    return re.sub(r"\s+", "", t or u"")


def valid(b):
    return bool(b) and len(b) == 4 and b[2] > b[0] and b[3] > b[1]


def center(b):
    return ((b[0] + b[2]) // 2, (b[1] + b[3]) // 2)


def in_list(b):
    """按钮是否真的落在当前屏弹窗任务列表的可视区。"""
    return (valid(b)
            and SAFE_TOP <= b[1]
            and b[3] <= SAFE_BOTTOM
            and b[0] >= BTN_MIN_X
            and b[2] <= BTN_MAX_X)


def is_button(n):
    """贴右 + 白名单文案 + 可视安全区。注意 XML 里 clickable=false。"""
    return norm(n.get("text")) in BTN_TEXTS and in_list(n.get("bounds"))


def classify(nodes):
    """页面状态：verify / panel / float / center / tejia / home / other。"""
    if not nodes:
        return "other"
    texts = set(norm(n.get("text")) for n in nodes)
    descs = set(norm(n.get("desc")) for n in nodes)
    blob = texts | descs
    if any(v in t for t in blob for v in VERIFY_MARKS):
        return "verify"
    # 弹窗是第一优先级：它叠在 center 之上，两页节点同时存在
    if any(PANEL_MARK in t or PANEL_SUB_MARK in t for t in texts):
        return "panel"
    if any(any(f in t for f in FLOAT_MARKS) for t in texts):
        return "float"
    # 落地页必须在 center 之前判：H5 会在同一个 WebView 里换页，
    # 商品页那帧的 XML 还残留着【领券中心】标题，只按 center 判会一直
    # 对着商品卡点【领更多】的坐标（真机 18:0x 那次就是这么空转的）。
    if any(any(m in t for m in LANDING_MARKS) for t in texts):
        return "landing"
    if any(norm(n.get("text")) == TAB_COUPON and n.get("class") == "WebView"
           for n in nodes):
        return "center"
    selected_tejia = any(TAB_TEJIA in d and u"已选中" in d for d in descs)
    selected_home = any(u"首页" in d and u"已选中" in d for d in descs)
    if selected_tejia and TAB_COUPON in texts:
        return "tejia"
    if selected_home:
        return "home"
    return "other"


def reward_of(nodes, b):
    """给按钮配同一行的【完成得Y京豆】奖励小字，返回 (京豆数, 原文)。"""
    cands = []
    for n in nodes:
        t = norm(n.get("text"))
        m = REWARD_RE.search(t)
        if not m:
            continue
        nb = n.get("bounds")
        if not valid(nb) or not (LEFT_MIN_X <= nb[0] <= LEFT_MAX_X):
            continue
        dy = nb[1] - b[1]
        if not (10 <= dy <= REWARD_MAX_DY):
            continue
        cands.append((abs(dy - 40), int(m.group(1)), t))
    if not cands:
        return (0, u"")
    cands.sort()
    return (cands[0][1], cands[0][2])


def row_title(nodes, b):
    """给右侧按钮配同行标题：带 (N/M) 进度的优先，其次纵向最近。"""
    cands = []
    for n in nodes:
        t = norm(n.get("text"))
        if not t or len(t) < 3 or t in ALL_BTN_TEXTS:
            continue
        if PRICE_RE.search(t) or REWARD_RE.search(t):
            continue
        nb = n.get("bounds")
        if not valid(nb) or not (LEFT_MIN_X <= nb[0] <= LEFT_MAX_X):
            continue
        dy = b[1] - nb[1]
        if not (-5 <= dy <= TITLE_MAX_DY):
            continue
        prog = bool(PROGRESS_RE.search(t))
        cands.append((0 if prog else 1, abs(dy - 27), nb[0], -len(t), t))
    if not cands:
        return u""
    cands.sort()
    return cands[0][4]


def task_key(title, btn_text, reward_text=u""):
    """台账 key 去掉 (N/M) 进度，任务做完进度 +1 时仍是同一个任务。"""
    base = PROGRESS_RE.sub(u"", title or u"")
    if not base:
        base = u"#" + norm(reward_text)[:12]
    return base + u"|" + norm(btn_text)


def _rows_from_nodes(nodes):
    out = []
    for n in nodes:
        if not is_button(n):
            continue
        b = n["bounds"]
        beans, reward_text = reward_of(nodes, b)
        if beans <= 0:
            continue                     # 非「完成得Y京豆」的行一律排除
        title = row_title(nodes, b)
        out.append({
            "text": norm(n.get("text")),
            "title": title,
            "reward": reward_text,
            "beans": beans,
            "bounds": tuple(b),
            "center": center(b),
            "key": task_key(title, n.get("text"), reward_text),
        })
    out.sort(key=lambda r: (r["bounds"][1], r["bounds"][0]))
    return out


def parse_rows(nodes):
    """XML 通道：解析当前屏可做任务。"""
    if classify(nodes) != "panel":
        return []
    return _rows_from_nodes(nodes)


def ocr_panel_visible(items):
    return any(PANEL_MARK in norm(it.get("text")) for it in (items or []))


def parse_ocr_rows(items):
    """OCR 通道：把 OCR 块伪装成 nodes 后复用同一套配对逻辑。"""
    if not ocr_panel_visible(items):
        return []
    nodes = []
    for it in (items or []):
        b = (int(it["x1"]), int(it["y1"]), int(it["x2"]), int(it["y2"]))
        nodes.append({"text": it.get("text", ""), "bounds": b,
                      "clickable": True, "class": "TextView"})
    return _rows_from_nodes(nodes)


def panel_sig(nodes):
    """当前屏所有按钮的坐标签名，用于判「滑到底 / 回到顶部」。"""
    items = []
    for n in nodes:
        t = norm(n.get("text"))
        b = n.get("bounds")
        if t in ALL_BTN_TEXTS and valid(b) and b[0] >= BTN_MIN_X:
            items.append((t, tuple(b)))
    items.sort()
    return tuple(items)


def ocr_sig(items):
    out = []
    for it in (items or []):
        t = norm(it.get("text"))
        if t in ALL_BTN_TEXTS:
            out.append((t, int(it["x1"]), int(it["y1"])))
    out.sort()
    return tuple(out)


def dwell_seconds(rnd=None):
    """每次浏览停留 9~10 秒（原来 12~15，提速后按需求压到 9 秒起）。"""
    r = rnd if rnd is not None else random
    return r.randint(DWELL_MIN, DWELL_MAX)


def progress_total(title):
    """标题里 (N/M) 的 M；没有进度返回 0。

    真机实测【逛活动页面领京豆(2/5)】这类任务是分 5 步的：
    只按固定次数封顶会让它停在 (4/5)，与「做到列表没有可做任务为止」冲突。
    """
    m = PROGRESS_RE.search(title or u"")
    if not m:
        return 0
    nums = re.findall(r"\d+", m.group(0))
    return int(nums[1]) if len(nums) >= 2 else 0


def task_limit(row, max_per_task=MAX_PER_TASK):
    """单任务允许做的次数：带 (N/M) 进度的跟到 M/M，无进度的用默认上限。"""
    total = progress_total(row.get("title"))
    if total <= 0:
        return max_per_task
    return max(max_per_task, min(total, MAX_PER_TASK_PROGRESS))


def pick_target(rows, blacklist, attempts, max_per_task=MAX_PER_TASK):
    for r in rows:
        if r["key"] in blacklist:
            continue
        if attempts.get(r["key"], 0) >= task_limit(r, max_per_task):
            continue
        return r
    return None


def ocr_panel_items(img_path, box=None, scale=None):
    """只识别底部面板区域并降采样（真机实测 23.4s -> 10.0s）。

    面板是固定在屏幕下半部的半屏 sheet，整屏 OCR 有一半算力浪费在背景上；
    裁到 OCR_BOX 再缩到 0.45 倍后【做任务得奖励】【完成得2京豆】【去浏览】
    仍然能读全（28 块 vs 39 块文本，行配对结果一致）。
    返回的坐标已换算回全屏，调用方不需要知道裁剪/缩放。
    """
    if not img_path or not os.path.isfile(img_path):
        return None
    from PIL import Image
    from modules.task5_joy_trip import get_ocr

    im = Image.open(img_path)
    W, H = im.size
    box = box or OCR_BOX
    scale = OCR_SCALE if scale is None else float(scale)
    x0, y0 = max(0, int(box[0])), max(0, int(box[1]))
    x1, y1 = min(W, int(box[2])), min(H, int(box[3]))
    if x1 - x0 < 120 or y1 - y0 < 120:
        # 换了分辨率 / box 不合身：退回整屏下半部，别把整轮任务搞崩
        x0, y0, x1, y1 = 0, OCR_CROP_TOP, W, H
    crop = im.crop((x0, y0, x1, y1))
    if scale != 1.0:
        crop = crop.resize((max(1, int(crop.size[0] * scale)),
                            max(1, int(crop.size[1] * scale))))
    path = img_path[:-4] + "_panel.png"
    crop.save(path)

    res = get_ocr().ocr(path, cls=False)
    k = 1.0 / scale
    out = []
    for quad, (txt, _score) in (res[0] if res and res[0] else []):
        xs = [p[0] for p in quad]
        ys = [p[1] for p in quad]
        out.append({
            "text": (txt or "").strip(),
            "cx": int(sum(xs) / 4 * k) + x0,
            "cy": int(sum(ys) / 4 * k) + y0,
            "x1": int(min(xs) * k) + x0,
            "y1": int(min(ys) * k) + y0,
            "x2": int(max(xs) * k) + x0,
            "y2": int(max(ys) * k) + y0,
        })
    return out


def _default_shot(ui):
    """真机截图：删旧图再截，避免拿到上一帧。"""
    path = os.path.join(PROJECT_ROOT, "_coupon_screen.png")

# ---------------- 模块主体 ----------------
class Task14Coupons(BaseModule):
    name = u"领券中心·天天领豆"
    description = (u"京东首页切【特价】→【领券中心】→【天天领豆/领更多】"
                   u"→ 只做带【完成得Y京豆】且按钮为"
                   u"【去浏览/逛一逛/去完成】的任务，每次停留 9~10 秒，"
                   u"循环到列表没有可做任务为止")
    icon = u"\U0001F3AB"

    def __init__(self):
        self.ui = None
        self.log = None
        self.stop = None
        self.expect_panel = False       # 已点过入口，才允许 OCR 兜底
        self.ocr_budget = MAX_OCR
        self.ocr_grace = 0              # 点完入口后的 OCR 帧预算
        self.ocr_calls = 0
        self.panel_rows = None          # 上次成功解析出的面板行（复用来省 OCR）
        self.stale_uses = 0
        self._verify_hit = False
        self._last = None               # _do_one 最后一次真实读屏结果，交给主循环复用

    # ---------- 基础封装 ----------
    def _say(self, msg, level="INFO"):
        if self.log is not None:
            self.log.write_log(msg, level)

    def _stopped(self):
        return self.stop is not None and self.stop.is_set()

    def _revive(self):
        """uiautomator2 instrumentation 崩了（ECONNREFUSED 8201）：

        utils.driver.get_driver() 自带探活+重建 session，这里把新 driver
        塞回 ui，否则整轮任务会陪葬（真机 19:0x 那次就是这么挂的）。
        """
        try:
            d = driver.get_driver()
        except Exception as e:
            self._say(u"重连 Appium 失败：%s" % e, "WARN")
            return False
        self.ui.d = d
        self.ui.invalidate()
        self._say(u"Appium session 已重建，继续跑", "WARN")
        return True

    def _snap(self):
        self.ui.invalidate()
        xml = self.ui.source(fresh=True)
        if not xml and self._revive():
            xml = self.ui.source(fresh=True)
        nodes = self.ui.nodes(xml) if xml else []
        return classify(nodes), nodes

    def _remember(self, rows):
        """记住刚解析出的面板行：面板不进无障碍树，下次判定在场时直接复用。"""
        self.panel_rows = rows
        self.stale_uses = 0
        # 面板已经确认在了，别再为「确认它在」花第二次 10~15s 的 OCR
        self.ocr_grace = 0

    def _back(self, pause=1.5):
        try:
            self.ui.back(pause=pause)
        except Exception:
            if self._revive():
                try:
                    self.ui.back(pause=pause)
                except Exception:
                    pass

    def _tap(self, b, pause=1.2):
        # b 来自 XML/OCR，已经是设备像素，不再 sx/sy 缩放
        self.ui.adb_tap(center(b), pause=pause)

    def _find_text(self, text, nodes=None, x_min=None, x_max=None,
                   y_min=None, y_max=None, clickable=None, exact=True):
        if nodes is None:
            _, nodes = self._snap()
        want = norm(text)
        for n in nodes:
            t = norm(n.get("text"))
            hit = (t == want) if exact else (want in t)
            if not hit:
                continue
            if clickable is not None and bool(n.get("clickable")) != clickable:
                continue
            b = n.get("bounds")
            if not valid(b):
                continue
            if (x_min is not None and b[0] < x_min) or \
               (x_max is not None and b[0] > x_max) or \
               (y_min is not None and b[1] < y_min) or \
               (y_max is not None and b[1] > y_max):
                continue
            return tuple(b)
        return None

    def _clickable_container(self, b, nodes):
        """H5 文本节点常不可点，取包住它的最小可点容器。"""
        cx, cy = center(b)
        hits = []
        for n in nodes:
            nb = n.get("bounds")
            if not n.get("clickable") or not valid(nb):
                continue
            if not (nb[0] <= cx <= nb[2] and nb[1] <= cy <= nb[3]):
                continue
            w, h = nb[2] - nb[0], nb[3] - nb[1]
            if w <= 400 and h <= 400:
                hits.append((w * h, tuple(nb)))
        if not hits:
            return None
        hits.sort()
        return hits[0][1]

    def _scroll(self, spec, pause=1.3):
        x1, y1, x2, y2 = spec
        self.ui.adb_swipe(self.ui.sx(x1), self.ui.sy(y1),
                          self.ui.sx(x2), self.ui.sy(y2), SWIPE_MS)
        self.ui.sleep(pause)
        self.ui.invalidate()

    def _scroll_up(self):
        self.panel_scrolled = True
        self._scroll(SWIPE_UP)

    def _scroll_down(self):
        self._scroll(SWIPE_DOWN)

    # ---------- OCR 兜底 ----------
    def _need_ocr(self, state):
        # 盲帧实测 classify 仍是 center（背景 WebView 在，弹窗节点不在）。
        # 只有 center 才兜底：XML 已经能读到【做任务得奖励】时按钮一定也在，
        # 此时 rows=[] 就是真的没任务了，不必每帧都跑一次 OCR。
        # 一次 OCR 30~45s（实测），只在「刚点完入口」的 2 帧里给机会：
        # 否则状态机每轮探针都跑 OCR，10 分钟都走不完一步。
        return (self.expect_panel and self.ocr_grace > 0
                and self.ocr_budget > 0 and state == "center")

    def _ocr_items(self):
        self.ocr_calls += 1
        self.ocr_budget -= 1
        self.ocr_grace -= 1
        try:
            path = SHOT_HOOK(self.ui) if SHOT_HOOK else _default_shot(self.ui)
            if OCR_HOOK:
                return OCR_HOOK(path)
            try:
                items = ocr_panel_items(path)
                if not items:
                    # 刚点完入口那一帧常常是白屏/动画中，空白结果不算数
                    self.ui.sleep(0.8)
                    path = SHOT_HOOK(self.ui) if SHOT_HOOK else _default_shot(self.ui)
                    items = ocr_panel_items(path)
                return items
            except Exception:
                # 面板裁剪切歪了就退回整屏下半部，别让一次 OCR 失败毁掉整轮
                from modules.task5_joy_trip import ocr_items
                return ocr_items(path, crop_top=OCR_CROP_TOP)
        except Exception as e:
            self._say(u"OCR 兜底不可用：%s" % e, "WARN")
            return None

    def _probe(self, ocr_first=False, allow_stale=True):
        """统一读帧：XML 优先，弹窗节点缺失时 OCR 兜底。

        返回 {'state','rows','nodes','sig','src'}；rows 永远是「可做任务」。
        """
        if ocr_first and self._need_ocr("center"):
            # 刚点过入口：这个 H5 面板不进无障碍树，先截图 OCR 比先 dump XML 省 5~8s
            items = self._ocr_items()
            seen = ocr_panel_visible(items)
            self._say(u"OCR 兜底：%d 块文本，弹窗标记=%s，剩余预算 %d"
                      % (len(items or []), seen, self.ocr_budget), "DEBUG")
            if items is not None and seen:
                rows = parse_ocr_rows(items)
                self._remember(rows)
                self._say(u"OCR 解析出 %d 行任务，前几行：%s"
                          % (len(rows), u" | ".join(
                              u"%s/%s" % (r.get("title"), r.get("text"))
                              for r in rows[:5]) or u"无"), "DEBUG")
                if not rows:
                    self._say(u"OCR 文本样本：%s"
                              % u" / ".join(i["text"][:12] for i in items[:12]),
                              "DEBUG")
                return {"state": "panel", "rows": rows,
                        "nodes": [], "sig": ocr_sig(items), "src": "ocr"}
            state, nodes = self._snap()
            return {"state": state, "rows": [], "nodes": nodes,
                    "sig": panel_sig(nodes), "src": "xml"}
        state, nodes = self._snap()
        sig = panel_sig(nodes)
        if state == "panel":
            rows = parse_rows(nodes)
            self._remember(rows)
            return {"state": "panel", "rows": rows,
                    "nodes": nodes, "sig": sig, "src": "xml"}
        if self._need_ocr(state):
            items = self._ocr_items()
            seen = ocr_panel_visible(items)
            self._say(u"OCR 兜底：%d 块文本，弹窗标记=%s，剩余预算 %d"
                      % (len(items or []), seen, self.ocr_budget), "DEBUG")
            if items is not None and seen:
                rows = parse_ocr_rows(items)
                self._remember(rows)
                self._say(u"OCR 解析出 %d 行任务，前几行：%s"
                          % (len(rows), u" | ".join(
                              u"%s/%s" % (r.get("title"), r.get("text"))
                              for r in rows[:5]) or u"无"), "DEBUG")
                if not rows:
                    self._say(u"OCR 文本样本：%s"
                              % u" / ".join(i["text"][:12] for i in items[:12]),
                              "DEBUG")
                return {"state": "panel", "rows": rows,
                        "nodes": nodes, "sig": ocr_sig(items), "src": "ocr"}
        if (allow_stale and self.expect_panel and state == "center"
                and self.panel_rows is not None
                and self.stale_uses < MAX_STALE):
            # 面板不进无障碍树：点开过、当前仍是领券中心底页(center)、上次的行
            # 还在，就认为面板仍在。否则每次 back 回来都要再花 10s OCR 才接得上。
            self.stale_uses += 1
            self._say(u"探针：面板缓存命中（第 %d 次），%d 行"
                      % (self.stale_uses, len(self.panel_rows or [])), "DEBUG")
            return {"state": "panel", "rows": self.panel_rows,
                    "nodes": nodes, "sig": sig, "src": "stale"}
        return {"state": state, "rows": [], "nodes": nodes,
                "sig": sig, "src": "xml"}

    # ---------- 入口链路 ----------
    def _clear_float(self):
        """关掉「恭喜获得 ¥15」浮层：按参考坐标点 X，再 back 兜底。"""
        for k in range(3):
            state, _ = self._snap()
            if state != "float":
                return True
            x, y = FLOAT_CLOSE_XY
            self.ui.adb_tap((self.ui.sx(x), self.ui.sy(y)), pause=1.2)
            self.ui.sleep(1.2)
            state2, _ = self._snap()
            if state2 != "float":
                self._say(u"浮层已关闭（点 X，第 %d 次）" % (k + 1))
                return True
            self._back(pause=1.2)
            self.ui.sleep(1.0)
        self._say(u"浮层未能关闭，交给入口状态机恢复", "WARN")
        return self._snap()[0] != "float"

    def _top_tab(self, nodes):
        return self._find_text(TAB_TEJIA, nodes, x_max=250, y_max=260)

    def _coupon_tab(self, nodes):
        b = self._find_text(TAB_COUPON, nodes, x_min=700,
                            y_min=300, y_max=650)
        if not b:
            return None
        return self._clickable_container(b, nodes) or b

    def _find_entry(self, nodes):
        """找【天天领豆】卡片：整卡是 Button，文案是【更多任务】/【领更多】。

        y 不要再卡死 600~1400：领券中心页偶发多插一行金刚区或更高的券 banner，
        实测同一天那张卡被顶到 y≈1685，卡死区间会让状态机静默空转。
        """
        for want in ENTRY_TEXTS:
            b = self._find_text(want, nodes, y_min=300, x_max=460,
                                clickable=True)
            if b:
                return b
        for want in ENTRY_TEXTS:
            b = self._find_text(want, nodes, y_min=300, x_max=460,
                                clickable=False)
            if b:
                return self._clickable_container(b, nodes) or b
        return None

    def _tap_entry(self, b):
        """整张卡片是一个 Button，【领更多】在底部，点下缘更稳。"""
        x = (b[0] + b[2]) // 2
        y = b[3] - 48
        if y <= b[1]:
            y = (b[1] + b[3]) // 2
        self.ui.adb_tap((x, y), pause=1.2)

    def _reach_center(self):
        """状态机：不管当前停在哪，用最短路径回到【领券中心】底页。"""
        for step in range(MAX_ENTER_STEPS):
            if self._stopped():
                return False
            state, nodes = self._snap()
            if state in ("center", "panel"):
                return True
            if state == "verify":
                self._say(u"停在【京东验证】风控页，需人工点掉", "WARN")
                return False
            if state == "float":
                self._clear_float()
                continue
            if state == "landing":
                self._say(u"停在商品落地页，back 回频道页")
                self._back(pause=1.5)
                self.ui.sleep(1.0)
                continue
            if state == "home":
                b = self._top_tab(nodes)
                if b:
                    self._say(u"点顶部一级 tab【特价】%s" % (b,))
                    self._tap(b, pause=2.0)
                    self.ui.sleep(1.5)
                    continue
            elif state == "tejia":
                b = self._coupon_tab(nodes)
                if b:
                    self._say(u"点二级 tab【领券中心】%s" % (b,))
                    self._tap(b, pause=2.0)
                    self.ui.sleep(1.8)
                    continue
            self.ui.sleep(1.0)
            if step % 3 == 2:
                self._back(pause=1.2)
        self._say(u"连续 %d 步没回到【领券中心】" % MAX_ENTER_STEPS, "WARN")
        return False

    def _open_panel(self):
        """保证【做任务得奖励】面板在场；XML 盲帧由 _probe 的 OCR 兜底识别。"""
        last_entry, same_entry = None, 0
        for k in range(MAX_OPEN_STEPS):
            if self._stopped():
                return False
            # 这里必须真实探：走缓存会把「面板其实已经关了」当成还在，
            # 于是整轮任务一个都没做就收尾（盲帧场景就是这么挂的）
            p = self._probe(ocr_first=(k > 0), allow_stale=False)
            state = p["state"]
            self._say(u"入口第 %d 步：state=%s src=%s rows=%d"
                      % (k + 1, state, p["src"], len(p["rows"])), "DEBUG")
            if state == "panel":
                return True
            if state == "verify":
                self._say(u"停在【京东验证】风控页，停止", "WARN")
                return False
            if state == "float":
                self._clear_float()
                continue
            if state in ("home", "tejia", "landing", "other"):
                if not self._reach_center():
                    return False
                continue
            # center：点【天天领豆 - 领更多 / 更多任务】
            b = self._find_entry(p["nodes"])
            if not b:
                if k % 3 == 2:
                    self._say(u"第 %d 次没找到【天天领豆/领更多】入口，"
                              u"等页面稳定" % (k + 1), "WARN")
                self.ui.sleep(1.2)
                continue
            if b == last_entry:
                same_entry += 1
            else:
                last_entry, same_entry = b, 1
            if same_entry >= 3:
                self._say(u"同一处【领更多】连点 3 次都没弹窗，"
                          u"先 back 出嵌套页再重进", "WARN")
                self._back(pause=1.5)
                self.ui.sleep(1.0)
                last_entry, same_entry = None, 0
                continue
            self._say(u"点【领更多/更多任务】%s（第 %d 次）" % (b, k + 1))
            self._tap_entry(b)
            self.expect_panel = True
            self.ocr_grace = OCR_GRACE      # 只给接下来 2 帧 OCR 机会
            self.panel_rows = None          # 刚换了面板，旧行作废
            self.stale_uses = 0
            self.ui.sleep(1.8)
        self._say(u"连续 %d 次没打开任务弹窗" % MAX_OPEN_STEPS, "WARN")
        return False

    def _scroll_to_top(self):
        """回顶只需按钮坐标签名，直接读 XML，别在这里耗 OCR 预算。

        面板是底部半屏 sheet：还没滑过就别动它——向下拖会把面板直接关掉
        （真机 19:2x 那次 scroll_to_top 把刚开的弹窗拖没了，之后一直在
        「盲帧 → 重点入口 → 再盲帧」里打转）。
        """
        if not getattr(self, "panel_scrolled", False):
            return
        same = 0
        self._say(u"先把任务列表滑回顶部再扫")
        for _ in range(MAX_TO_TOP):
            if self._stopped():
                return
            _, nodes = self._snap()
            sig = panel_sig(nodes)
            self._scroll_down()
            _, nodes2 = self._snap()
            if panel_sig(nodes2) == sig:
                same += 1
                if same >= 2:
                    return
            else:
                same = 0

    # ---------- 做任务 ----------
    def _do_one(self, row):
        """点按钮 → 确认离开弹窗 → 停留 9~10s → back 一次回弹窗。"""
        self._last = None
        pt = row.get("center") or center(row["bounds"])
        self.ui.adb_tap(pt, pause=0.45)
        self.ui.sleep(0.9)
        # allow_stale=False：这里若用缓存把「页面已跳走」误判成「面板还在」，
        # 第二次点击就会落在商品页上（可能误碰加购），宁可多走一次 back。
        p = self._probe(allow_stale=False)
        if p["state"] == "verify":
            self._verify_hit = True
            self._say(u"停在【京东验证】风控页，停止", "WARN")
            return False
        if p["state"] in ("panel", "center"):
            # 面板不进无障碍树时，面板开着 state 也是 center：再等一帧看页面跳没跳。
            # 跳走了才是真开始浏览；两帧都没动就是点击被吞，直接失败——
            # 不重复点，否则第二次点击可能落在商品页的按钮上。
            self.ui.sleep(1.2)
            p = self._probe(allow_stale=False)
            if p["state"] == "verify":
                self._verify_hit = True
                self._say(u"停在【京东验证】风控页，停止", "WARN")
                return False
            if p["state"] in ("panel", "center"):
                self._say(u"点了按钮页面没动，本次点击可能被吞", "WARN")
                return False
        secs = dwell_seconds()
        self._say(u"任务【%s】%s 按钮【%s】%s，停留 %ds"
                  % (row["title"] or row["key"], row["reward"],
                     row["text"], pt, secs))
        if not self.ui.sleep(secs):
            return False
        self._back(pause=1.2)
        p = self._probe()          # 只有这里允许走缓存：刚 back，面板大概率还在
        for _ in range(2):
            if p["state"] != "landing":
                break
            self._say(u"back 一次还停在商品落地页，再 back 一次")
            self._back(pause=1.0)
            p = self._probe(allow_stale=False)
        if p["state"] != "panel":
            for _ in range(2):
                self.ui.sleep(0.7)
                p = self._probe(allow_stale=False)
                if p["state"] == "panel":
                    break
            if p["state"] != "panel":
                self._say(u"浏览后没回到任务弹窗，重建入口", "WARN")
                self._open_panel()
                return True
        # 交给主循环复用：面板不进无障碍树，主循环拿不到就得重开弹窗再 OCR 一次，
        # 那是每个任务多花 10~15s 的主因。stale 里的行已由主循环按 key 摘掉。
        self._last = p
        return True

    def _do_tasks(self):
        total = 0
        attempts = {}
        for cycle in range(MAX_CYCLES):
            if self._stopped() or total >= MAX_TASKS:
                break
            self._say(u"===== 第 %d 轮：扫描并做任务 =====" % (cycle + 1))
            # 第 1 轮 run() 已经把弹窗开好了，再开一次等于白花 10~20s 的 OCR
            if cycle > 0 and not self._open_panel():
                break
            self._scroll_to_top()
            done = 0
            blacklist = set()
            failures = {}
            read_fails = 0
            stagnant = 0
            scrolls = 0
            frame_rows = None     # 当前屏解析结果：做完一个任务直接复用，
            frame_sig = ()        # 省掉一次 dump(3s) + 一次 OCR(10s)
            for step in range(MAX_STEPS):
                if self._stopped() or total >= MAX_TASKS:
                    break
                if frame_rows is None:
                    # 允许走面板缓存：否则每次都要重开弹窗 + 再 OCR 一遍
                    p = self._probe()
                    if not p["nodes"] and p["src"] == "xml":
                        read_fails += 1
                        if read_fails >= MAX_READ_FAILS:
                            self._say(u"连续 %d 次读不到页面，收尾" % read_fails,
                                      "WARN")
                            break
                        self.ui.sleep(0.8)
                        continue
                    read_fails = 0
                    if p["state"] == "verify":
                        self._say(u"停在【京东验证】风控页，停止", "WARN")
                        return total
                    if p["state"] != "panel":
                        if not self._open_panel():
                            break
                        p = self._probe(allow_stale=False)
                        if p["state"] != "panel":
                            break
                    frame_rows = p["rows"]
                    frame_sig = p["sig"]
                pick = pick_target(frame_rows, blacklist, attempts)
                if pick:
                    key = pick["key"]
                    ok = self._do_one(pick)
                    attempts[key] = attempts.get(key, 0) + 1
                    if ok:
                        total += 1
                        done += 1
                        failures[key] = 0
                        self._say(u"浏览完成 #%d / 本轮 #%d【%s】得 %d 京豆"
                                  % (total, done, pick["title"] or key,
                                     pick["beans"]))
                        left = task_limit(pick) - attempts[key]
                        if left > 0:
                            self._say(u"  同一任务还需 %d 次才做满进度" % left)
                        # _do_one 刚真实读过这一屏，直接拿来挑下一个（省一次 dump）；
                        # 但要把刚做完的行摘掉，否则会对着「已完成」再点一次
                        rows = (self._last or {}).get("rows") or []
                        frame_rows = [r for r in rows
                                      if r.get("key") != key] or None
                    else:
                        if self._verify_hit:
                            self._say(u"撞上【京东验证】风控页，整轮停止", "WARN")
                            return total
                        n = failures.get(key, 0) + 1
                        failures[key] = n
                        if n >= MAX_RETRY:
                            blacklist.add(key)
                            self._say(u"任务【%s】连续 %d 次没成功，本轮拉黑"
                                      % (key, n), "WARN")
                        frame_rows = None   # 没成功就老实重读这一屏
                    continue               # 下一轮直接用同一屏的解析结果
                self._scroll_up()
                frame_rows = None          # 滑过之后坐标全变了，必须重读
                p2 = self._probe()
                if p2["sig"] == frame_sig:
                    stagnant += 1
                else:
                    stagnant = 0
                scrolls += 1
                if stagnant >= 2 or scrolls >= MAX_SCROLLS:
                    self._say(u"本屏无【完成得Y京豆】可做任务，列表已扫到底")
                    break
                frame_rows = p2["rows"]
                frame_sig = p2["sig"]
            self._say(u"第 %d 轮完成浏览 %d 次（累计 %d 次）"
                      % (cycle + 1, done, total))
            if done == 0:
                self._say(u"任务列表已没有可做任务，结束")
                break
        if total >= MAX_TASKS:
            self._say(u"已达总浏览次数上限 %d，收尾" % MAX_TASKS, "WARN")
        return total

    # ---------- 入口 ----------
    def run(self, log, stop_event):
        ui = JDUI(driver.get_driver(), log, stop_event)
        self.ui = ui
        self.log = log
        self.stop = stop_event
        self.expect_panel = False
        self.ocr_budget = MAX_OCR
        self.ocr_grace = 0
        self.ocr_calls = 0
        self.panel_scrolled = False
        self.panel_rows = None
        self.stale_uses = 0
        self._verify_hit = False
        self._last = None
        self._say(u"设备分辨率 %dx%d" % (ui.W, ui.H))
        if not ui.restart_to_home():
            self._say(u"冷启动回首页失败，走 goto_home", "WARN")
            ui.goto_home()
        if not self._reach_center():
            log.done(u"领券中心 未完成（没能进入【领券中心】）")
            return
        if not self._open_panel():
            log.done(u"领券中心 未完成（没能打开任务弹窗）")
            return
        total = self._do_tasks()
        log.done(u"领券中心 完成 %d 次【完成得Y京豆】浏览任务（OCR 兜底 %d 次）"
                 % (total, self.ocr_calls))
